2. POP Operation — Pseudocode
PROCEDURE POP()

    IF top = -1 THEN
        DISPLAY "Stack is empty"
        STOP
    END IF

    value = elements[top]
    top = top - 1

    RETURN value

END PROCEDURE

4. POSTFIX EVALUATION — Pseudocode
FUNCTION EVALUATE_POSTFIX(expression)

    IF expression is empty THEN
        DISPLAY "Expression cannot be empty"
        STOP
    END IF

    tokens = SPLIT expression by spaces

    CREATE empty stack

    FOR EACH token IN tokens

        IF token is an operator THEN

            IF stack is empty THEN
                DISPLAY "Invalid expression"
                STOP
            END IF

            operand2 = POP stack

            IF stack is empty THEN
                DISPLAY "Invalid expression"
                STOP
            END IF

            operand1 = POP stack

            result = PERFORM_OPERATION(
                        operand1,
                        operand2,
                        token
                     )

            PUSH result onto stack

        ELSE

            IF token is a valid number THEN
                PUSH token onto stack
            ELSE
                DISPLAY "Invalid token"
                STOP
            END IF

        END IF

    END FOR

    finalResult = POP stack

    IF stack is not empty THEN
        DISPLAY "Expression is malformed"
        STOP
    END IF

    RETURN finalResult

END FUNCTION

6. Daily Statistics — Pseudocode

PROCEDURE DISPLAY_DAILY_STATISTICS(serviceTimes)

    IF serviceTimes is empty THEN
        DISPLAY "No student tracking records found"
        STOP
    END IF

    totalStudents = LENGTH(serviceTimes)
    totalServiceTime = 0

    highestServiceTime = serviceTimes[0]
    lowestServiceTime = serviceTimes[0]

    studentsOver10Min = 0

    FOR each time IN serviceTimes

        totalServiceTime = totalServiceTime + time

        IF time > highestServiceTime THEN
            highestServiceTime = time
        END IF

        IF time < lowestServiceTime THEN
            lowestServiceTime = time
        END IF

        IF time > 10 THEN
            studentsOver10Min = studentsOver10Min + 1
        END IF

    END FOR

    averageServiceTime =
        totalServiceTime / totalStudents

    DISPLAY totalStudents
    DISPLAY totalServiceTime
    DISPLAY averageServiceTime
    DISPLAY highestServiceTime
    DISPLAY lowestServiceTime
    DISPLAY studentsOver10Min

END PROCEDURE

7. Merge Sort — Pseudocode
PROCEDURE MERGE_SORT(array, left, right)

    IF left < right THEN

        middle = (left + right) / 2

        MERGE_SORT(array, left, middle)

        MERGE_SORT(array, middle + 1, right)

        MERGE(array, left, middle, right)

    END IF

END PROCEDURE
