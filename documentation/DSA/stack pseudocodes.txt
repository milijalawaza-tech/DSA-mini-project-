PROCEDURE push(value)
    IF top EQUALS capacity - 1 THEN
        THROW StackOverflowError
    ENDIF
    INCREMENT top
    elements[top] = value

PROCEDURE pop()
    IF top EQUALS -1 THEN
        THROW EmptyStackException
    ENDIF
    value = elements[top]
    DECREMENT top
    RETURN value
ENDPROCEDURE
